# i18n Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Add full i18n support (Norwegian + English) to the portfolio with auto-detection and manual toggle.

**Architecture:** A lightweight `i18n.js` module with `t()` function for resolving translations. Static HTML strings use `data-i18n` attributes; JS-rendered content uses `{ en, no }` objects in data arrays. Language toggle in the header. Re-render on language change.

**Tech Stack:** Vanilla JS ES modules, no i18n libraries.

### Task 1: Create i18n core module

**Files:**
- Create: `scripts/i18n.js`

**Step 1: Create the i18n module**

```js
// scripts/i18n.js
import en from './translations/en.js';
import no from './translations/no.js';

const translations = { en, no };
let currentLang = 'en';
const listeners = [];

/** Resolve a dot-path key from a translations object. */
function lookupKey(obj, key) {
  return key.split('.').reduce((o, k) => o?.[k], obj);
}

/** Detect language from localStorage or navigator. */
export function getLang() {
  const stored = localStorage.getItem('lang');
  if (stored === 'en' || stored === 'no') return stored;
  const nav = (navigator.language || '').toLowerCase();
  return nav.startsWith('nb') || nav.startsWith('nn') || nav.startsWith('no') ? 'no' : 'en';
}

/**
 * Translate a value.
 * - String dot-path key (e.g. 'nav.projects') → looks up in translations
 * - { en, no } object → returns value for current language
 * - Plain string with no matching key → returns as-is
 */
export function t(keyOrValue, params = {}) {
  let result;

  if (keyOrValue && typeof keyOrValue === 'object' && ('en' in keyOrValue || 'no' in keyOrValue)) {
    result = keyOrValue[currentLang] ?? keyOrValue.en ?? '';
  } else if (typeof keyOrValue === 'string') {
    const looked = lookupKey(translations[currentLang], keyOrValue);
    result = looked !== undefined ? looked : keyOrValue;
  } else {
    return String(keyOrValue ?? '');
  }

  // Simple interpolation: replace {key} with params[key]
  for (const [k, v] of Object.entries(params)) {
    result = result.replaceAll(`{${k}}`, v);
  }
  return result;
}

/** Get full translations object for current language (for arrays like footer messages). */
export function getTranslations() {
  return translations[currentLang];
}

/** Update all DOM elements with data-i18n attributes. */
function updateDOM() {
  document.documentElement.lang = currentLang;

  document.querySelectorAll('[data-i18n]').forEach((el) => {
    const key = el.dataset.i18n;
    const value = lookupKey(translations[currentLang], key);
    if (value !== undefined) el.textContent = value;
  });

  document.querySelectorAll('[data-i18n-aria]').forEach((el) => {
    const key = el.dataset.i18nAria;
    const value = lookupKey(translations[currentLang], key);
    if (value !== undefined) el.setAttribute('aria-label', value);
  });

  document.querySelectorAll('[data-i18n-alt]').forEach((el) => {
    const key = el.dataset.i18nAlt;
    const value = lookupKey(translations[currentLang], key);
    if (value !== undefined) el.alt = value;
  });

  // Update lang toggle button
  const toggle = document.querySelector('[data-lang-toggle]');
  if (toggle) {
    toggle.querySelectorAll('.lang-toggle__option').forEach((opt) => {
      opt.classList.toggle('lang-toggle__option--active', opt.dataset.lang === currentLang);
    });
  }
}

/** Set language, update DOM, notify listeners. */
export function setLang(lang) {
  if (lang !== 'en' && lang !== 'no') return;
  currentLang = lang;
  localStorage.setItem('lang', lang);
  updateDOM();
  listeners.forEach((fn) => fn(lang));
}

/** Register a callback for language changes (used for re-rendering JS content). */
export function onLangChange(callback) {
  listeners.push(callback);
}

/** Initialize i18n — call once on page load. */
export function initI18n() {
  currentLang = getLang();
  updateDOM();

  // Language toggle click handler
  document.querySelector('[data-lang-toggle]')?.addEventListener('click', () => {
    setLang(currentLang === 'en' ? 'no' : 'en');
  });
}
```
